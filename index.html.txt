<script>
const APP_CONFIG = {
    PURGE_DAYS: 15,
    DEFAULT_PASSWORD: "ward@2024",
    ADMIN_USER: "admin",
    ADMIN_PASS: "admin@2024"
};

// Sample drain data: 34 wards, 3-5 drains each
const DRAIN_DATA = [];
for (let w = 1; w <= 34; w++) {
    const drainCount = 3 + (w % 3);
    for (let d = 1; d <= drainCount; d++) {
        DRAIN_DATA.push({
            drain_id: `W${w.toString().padStart(2, '0')}_D${d}`,
            ward_no: w,
            location: `Ward ${w} - ${['Main Road', 'Market Street', 'Temple Road', 'School Lane', 'Park Avenue'][d - 1] || 'Cross Street ' + d}`
        });
    }
}

class AntiLarvaTracker {
    constructor() {
        this.currentUser = null;
        this.currentStream = null;
        this.currentGPS = null;
        this.watchId = null;
        this.charts = {};
        this.init();
    }

    init() {
        this.purgeOldEntries();
        this.initializeDemoUsers();
        this.checkAuth();
    }

    initializeDemoUsers() {
        if (!localStorage.getItem('users_initialized')) {
            const users = {};
            for (let i = 1; i <= 34; i++) {
                users[`ward${i}`] = { password: APP_CONFIG.DEFAULT_PASSWORD, ward_no: i, role: 'ward' };
            }
            users[APP_CONFIG.ADMIN_USER] = { password: APP_CONFIG.ADMIN_PASS, role: 'admin' };
            localStorage.setItem('users', JSON.stringify(users));
            localStorage.setItem('users_initialized', 'true');
        }
    }

    purgeOldEntries() {
        const entries = this.getEntries();
        const cutoff = Date.now() - (APP_CONFIG.PURGE_DAYS * 24 * 60 * 60 * 1000);
        const filtered = entries.filter(e => new Date(e.timestamp).getTime() > cutoff);
        if (filtered.length !== entries.length) {
            localStorage.setItem('spray_entries', JSON.stringify(filtered));
        }
    }

    checkAuth() {
        const session = localStorage.getItem('current_session');
        if (session) {
            this.currentUser = JSON.parse(session);
            this.currentUser.role === 'admin' ? this.renderAdminDashboard() : this.renderWardDashboard();
        } else {
            this.renderLogin();
        }
    }

    login(username, password) {
        const users = JSON.parse(localStorage.getItem('users') || '{}');
        const user = users[username];
        if (user && user.password === password) {
            this.currentUser = { username, ...user };
            localStorage.setItem('current_session', JSON.stringify(this.currentUser));
            this.showToast('Login successful', 'success');
            user.role === 'admin' ? this.renderAdminDashboard() : this.renderWardDashboard();
            return true;
        }
        this.showToast('Invalid credentials', 'error');
        return false;
    }

    logout() {
        this.stopCamera();
        this.stopGPSWatch();
        localStorage.removeItem('current_session');
        this.currentUser = null;
        this.renderLogin();
    }

    getEntries() {
        return JSON.parse(localStorage.getItem('spray_entries') || '[]');
    }

    saveEntry(entry) {
        const entries = this.getEntries();
        entries.push({
            id: 'E' + Date.now() + Math.random().toString(36).substr(2, 9),
            ...entry
        });
        localStorage.setItem('spray_entries', JSON.stringify(entries));
    }

    showToast(message, type = 'info') {
        const colors = {
            success: 'bg-emerald-500',
            error: 'bg-red-500',
            warning: 'bg-amber-500',
            info: 'bg-sky-500'
        };
        const toast = document.createElement('div');
        toast.className = `toast ${colors[type]} text-white px-6 py-3 rounded-lg shadow-lg flex items-center gap-3`;
        toast.innerHTML = `
            <span>${message}</span>
            <button onclick="this.parentElement.remove()" class="ml-2 text-white/80 hover:text-white">✕</button>
        `;
        document.getElementById('toastContainer').appendChild(toast);
        setTimeout(() => toast.remove(), 4000);
    }

    async startCamera() {
        try {
            this.stopCamera();
            const stream = await navigator.mediaDevices.getUserMedia({
                video: { facingMode: 'environment', width: { ideal: 1280 }, height: { ideal: 720 } }
            });
            this.currentStream = stream;
            const video = document.getElementById('video');
            if (video) {
                video.srcObject = stream;
                document.getElementById('cameraPlaceholder')?.classList.add('hidden');
                document.getElementById('captureBtn')?.classList.remove('hidden');
            }
            return true;
        } catch (err) {
            this.showToast('Camera access denied. Please enable camera permissions.', 'error');
            return false;
        }
    }

    stopCamera() {
        if (this.currentStream) {
            this.currentStream.getTracks().forEach(track => track.stop());
            this.currentStream = null;
        }
    }

    capturePhoto() {
        const video = document.getElementById('video');
        const canvas = document.createElement('canvas');
        canvas.width = video.videoWidth;
        canvas.height = video.videoHeight;
        const ctx = canvas.getContext('2d');
        ctx.drawImage(video, 0, 0);
        const photo = canvas.toDataURL('image/jpeg', 0.8);
        this.stopCamera();
        return photo;
    }

    startGPSWatch() {
        if (!navigator.geolocation) {
            this.showToast('GPS not supported on this device', 'error');
            return;
        }
        document.getElementById('gpsStatus').innerHTML = '<div class="loader"></div><span class="ml-2">Acquiring GPS...</span>';
        this.watchId = navigator.geolocation.watchPosition(
            (pos) => {
                this.currentGPS = {
                    lat: pos.coords.latitude,
                    lng: pos.coords.longitude,
                    accuracy: pos.coords.accuracy,
                    timestamp: pos.timestamp
                };
                this.updateGPSDisplay();
            },
            (err) => {
                document.getElementById('gpsStatus').innerHTML = `
                    <div class="flex items-center text-red-600">
                        <svg class="w-5 h-5 mr-2" fill="currentColor" viewBox="0 0 20 20"><path fill-rule="evenodd" d="M10 18a8 8 0 100-16 8 8 0 000 16zM8.707 7.293a1 1 0 00-1.414 1.414L8.586 10l-1.293 1.293a1 1 0 101.414 1.414L10 11.414l1.293 1.293a1 1 0 001.414-1.414L11.414 10l1.293-1.293a1 1 0 00-1.414-1.414L10 8.586 8.707 7.293z" clip-rule="evenodd"/></svg>
                        <span>GPS Error: ${err.message}</span>
                    </div>
                `;
                document.getElementById('submitBtn').disabled = true;
            },
            { enableHighAccuracy: true, maximumAge: 0, timeout: 10000 }
        );
    }

    stopGPSWatch() {
        if (this.watchId) {
            navigator.geolocation.clearWatch(this.watchId);
            this.watchId = null;
        }
    }

    updateGPSDisplay() {
        if (!this.currentGPS) return;
        const accuracyColor = this.currentGPS.accuracy < 20 ? 'text-green-600' : 
                             this.currentGPS.accuracy < 50 ? 'text-amber-600' : 'text-red-600';
        document.getElementById('gpsStatus').innerHTML = `
            <div class="space-y-2">
                <div class="flex items-center text-green-600">
                    <svg class="w-5 h-5 mr-2" fill="currentColor" viewBox="0 0 20 20"><path fill-rule="evenodd" d="M5.05 4.05a7 7 0 119.9 9.9L10 18.9l-4.95-4.95a7 7 0 010-9.9zM10 11a2 2 0 100-4 2 2 0 000 4z" clip-rule="evenodd"/></svg>
                    <span class="font-medium">GPS Active</span>
                </div>
                <div class="text-sm text-gray-600 space-y-1">
                    <div>Lat: ${this.currentGPS.lat.toFixed(6)}</div>
                    <div>Lng: ${this.currentGPS.lng.toFixed(6)}</div>
                    <div class="${accuracyColor}">Accuracy: ±${Math.round(this.currentGPS.accuracy)}m</div>
                </div>
            </div>
        `;
        document.getElementById('submitBtn').disabled = false;
    }

    renderLogin() {
        document.getElementById('app').innerHTML = `
            <div class="min-h-screen flex items-center justify-center p-4">
                <div class="w-full max-w-md animate-in">
                    <div class="glass rounded-2xl shadow-2xl p-8 border border-white/20">
                        <div class="text-center mb-8">
                            <div class="inline-flex items-center justify-center w-16 h-16 bg-gradient-to-br from-sky-500 to-blue-600 rounded-2xl mb-4 shadow-lg">
                                <svg class="w-8 h-8 text-white" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M9 12l2 2 4-4m5.618-4.016A11.955 0 0112 2.944a11.955 11.955 0 01-8.618 3.04A12.02 12.02 0 003 9c0 5.591 3.824 10.29 9 11.622 5.176-1.332 9-6.03 9-11.622 0-1.042-.133-2.052-.382-3.016z"/></svg>
                            </div>
                            <h1 class="text-2xl font-bold text-gray-900 mb-2">Anti-Larva Tracker</h1>
                            <p class="text-gray-600 text-sm">Spray Work Monitoring System</p>
                        </div>
                        
                        <form id="loginForm" class="space-y-4">
                            <div>
                                <label class="block text-sm font-medium text-gray-700 mb-2">Username</label>
                                <input type="text" id="username" required class="w-full px-4 py-3 rounded-lg border border-gray-300 focus:ring-2 focus:ring-sky-500 focus:border-transparent transition" placeholder="ward1 or admin">
                            </div>
                            <div>
                                <label class="block text-sm font-medium text-gray-700 mb-2">Password</label>
                                <input type="password" id="password" required class="w-full px-4 py-3 rounded-lg border border-gray-300 focus:ring-2 focus:ring-sky-500 focus:border-transparent transition" placeholder="Enter password">
                            </div>

                                Sign In

                    </div>

                </div>

        document.getElementById('loginForm').addEventListener('submit', (e) => {
            e.preventDefault();
            const username = document.getElementById('username').value.trim();

            this.login(username, password);
        });
    }

                                        <div class="text-center">
                                            <svg class="w-16 h-16 mx-auto mb-2" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M3 9a2 2 0 012-2h.93a2 2 0 001.664-.89l.812-1.22A2 2 0 0110.07 4h3.86a2 2 0 011.664.89l.812 1.22A2 2 0 0018.07 7H19a2 2 0 012 2v9a2 2 0 01-2 2H5a2 2 0 01-2-2V9z"/><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M15 13a3 3 0 11-6 0 3 3 0 016 0z"/></svg>

    handleStartCamera() {
        this.startCamera().then(success => {
            if (success) {
                this.startGPSWatch();
                document.getElementById('startCameraBtn').classList.add('hidden');
                document.getElementById('video').classList.remove('hidden');
            }
        });
    }

    handleCapture() {
        const photo = this.capturePhoto();
        const canvas = document.getElementById('photoPreview');
        canvas.classList.remove('hidden');
        canvas.getContext('2d').drawImage(document.getElementById('video'), 0, 0);
        document.getElementById('video').classList.add('hidden');
        document.getElementById('captureBtn').classList.add('hidden');
        document.getElementById('retakeBtn').classList.remove('hidden');
        this.capturedPhoto = photo;
    }

    handleRetake() {
        this.capturedPhoto = null;
        document.getElementById('photoPreview').classList.add('hidden');
        document.getElementById('video').classList.remove('hidden');
        document.getElementById('captureBtn').classList.remove('hidden');
        document.getElementById('retakeBtn').classList.add('hidden');
        this.startCamera();
    }

    handleSubmit() {
        const drainId = document.getElementById('drainSelect').value;
        if (!drainId) {
            this.showToast('Please select a drain', 'warning');
            return;
        }
        if (!this.capturedPhoto) {
            this.showToast('Please capture a photo', 'warning');
            return;
        }
        if (!this.currentGPS) {
            this.showToast('GPS location required', 'warning');
            return;
        }
        if (this.currentGPS.accuracy > 100) {
            if (!confirm(`GPS accuracy is low (±${Math.round(this.currentGPS.accuracy)}m). Continue anyway?`)) {
                return;
            }
        }

        const drain = DRAIN_DATA.find(d => d.drain_id === drainId);
        const entry = {
            ward_no: this.currentUser.ward_no,
            drain_id: drainId,
            location_name: drain.location,
            photo_base64: this.capturedPhoto,
            gps_lat: this.currentGPS.lat,
            gps_lng: this.currentGPS.lng,
            gps_accuracy: this.currentGPS.accuracy,
            timestamp: new Date().toISOString(),
            submitted_by: this.currentUser.username
        };

        this.saveEntry(entry);
        this.showToast('Spray entry submitted successfully!', 'success');
        
        // Reset form
        document.getElementById('drainSelect').value = '';
        this.capturedPhoto = null;
        document.getElementById('photoPreview').classList.add('hidden');
        document.getElementById('retakeBtn').classList.add('hidden');
        document.getElementById('startCameraBtn').classList.remove('hidden');
        document.getElementById('cameraPlaceholder').classList.remove('hidden');
        this.stopGPSWatch();
        
        // Refresh view
        setTimeout(() => this.renderWardDashboard(), 1000);
    }

    renderAdminDashboard() {
        const entries = this.getEntries();
        const wardStats = this.calculateWardStats(entries);
        const totalDrains = DRAIN_DATA.length;
        const totalSprayed = entries.length;
        const overallCompletion = totalDrains > 0 ? Math.round((totalSprayed / totalDrains) * 100) : 0;

    calculateWardStats(entries) {
        const stats = [];
        for (let w = 1; w <= 34; w++) {
            const wardDrains = DRAIN_DATA.filter(d => d.ward_no === w);
            const wardEntries = entries.filter(e => e.ward_no === w);
            const sprayed = wardEntries.length;
            const total = wardDrains.length;
            stats.push({
                ward: w,
                totalDrains: total,
                sprayed: sprayed,
                pending: total - sprayed,
                completion: total > 0 ? Math.round((sprayed / total) * 100) : 0
            });
        }
        return stats;
    }

    renderWardChart(wardStats) {
        const ctx = document.getElementById('wardChart');
        if (!ctx) return;
        
        if (this.charts.wardChart) {
            this.charts.wardChart.destroy();
        }

        this.charts.wardChart = new Chart(ctx, {
            type: 'bar',
            data: {
                labels: wardStats.map(s => `W${s.ward}`),
                datasets: [{
                    label: 'Completion %',
                    data: wardStats.map(s => s.completion),
                    backgroundColor: wardStats.map(s => 
                        s.completion === 100 ? 'rgba(16, 185, 129, 0.8)' :
                        s.completion > 50 ? 'rgba(14, 165, 233, 0.8)' :
                        s.completion > 0 ? 'rgba(245, 158, 11, 0.8)' :
                        'rgba(156, 163, 175, 0.8)'
                    ),
                    borderColor: wardStats.map(s => 
                        s.completion === 100 ? 'rgb(16, 185, 129)' :
                        s.completion > 50 ? 'rgb(14, 165, 233)' :
                        s.completion > 0 ? 'rgb(245, 158, 11)' :
                        'rgb(156, 163, 175)'
                    ),
                    borderWidth: 1
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: true,
                plugins: {
                    legend: { display: false },
                    tooltip: {
                        callbacks: {
                            label: (context) => {
                                const stat = wardStats[context.dataIndex];
                                return [
                                    `Completion: ${context.parsed.y}%`,
                                    `Sprayed: ${stat.sprayed}/${stat.totalDrains}`
                                ];
                            }
                        }
                    }
                },
                scales: {
                    y: {
                        beginAtZero: true,
                        max: 100,
                        ticks: { callback: value => value + '%' }
                    }
                }
            }
        });
    }

    filterAdminView() {
        const wardFilter = document.getElementById('wardFilter').value;
        const dateFilter = document.getElementById('dateFilter').value;
        
        document.querySelectorAll('.ward-card').forEach(card => {
            const showWard = wardFilter === 'all' || card.dataset.ward === wardFilter;
            card.style.display = showWard ? 'block' : 'none';
        });

        document.querySelectorAll('.entry-item').forEach(item => {
            const showWard = wardFilter === 'all' || item.dataset.ward === wardFilter;
            const showDate = !dateFilter || item.dataset.date === dateFilter;
            item.style.display = showWard && showDate ? 'block' : 'none';
        });
    }
